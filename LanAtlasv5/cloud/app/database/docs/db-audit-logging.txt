
-- ---------------------------------------------------------------------------
-- audit_log
-- Immutable record of every human-initiated mutation to tenant-owned data
-- (device renamed/deleted/authorized, alert acknowledged/resolved, user role
-- changed, agent key rotated, etc.).
--
-- This is distinct from observations/alerts/analyst_notes, which are
-- machine-generated or optional. audit_log is mandatory for every mutation
-- an authenticated user makes through the dashboard.
--
-- OWASP A09 — Security Logging and Alerting Failures
-- OWASP ASVS V7 — Error Handling and Logging
-- ---------------------------------------------------------------------------

CREATE TABLE audit_log (
    id              BIGSERIAL       PRIMARY KEY,
    organization_id INTEGER         NOT NULL,
    user_id         INTEGER,                    -- who made the change
    action          VARCHAR(60)     NOT NULL,    -- e.g. 'device.rename'
    entity_type     VARCHAR(30)     NOT NULL,    -- 'device', 'alert', 'user', 'site', 'agent'
    entity_id       INTEGER,
    before_value    JSONB,                       -- state before the change (secrets redacted)
    after_value     JSONB,                       -- state after the change (secrets redacted)
    ip_address      VARCHAR(45),
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_audit_log_organization
        FOREIGN KEY (organization_id) REFERENCES organization(id)
        ON DELETE RESTRICT,

    CONSTRAINT fk_audit_log_user
        FOREIGN KEY (user_id) REFERENCES users(id)
        ON DELETE SET NULL
);

CREATE INDEX idx_audit_log_org_time ON audit_log (organization_id, created_at DESC);
CREATE INDEX idx_audit_log_entity   ON audit_log (entity_type, entity_id);

-- Immutability: once written, an audit row can never be altered or removed.
-- Enforced at the database level, not just by convention — even a
-- compromised application server with valid DB credentials cannot
-- tamper with the audit trail.
CREATE OR REPLACE FUNCTION fn_prevent_audit_log_mutation()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    RAISE EXCEPTION 'audit_log rows are immutable and cannot be updated or deleted';
END;
$$;

CREATE TRIGGER trg_audit_log_no_update
    BEFORE UPDATE ON audit_log
    FOR EACH ROW EXECUTE FUNCTION fn_prevent_audit_log_mutation();

CREATE TRIGGER trg_audit_log_no_delete
    BEFORE DELETE ON audit_log
    FOR EACH ROW EXECUTE FUNCTION fn_prevent_audit_log_mutation();
ENDOFFILE
echo "audit_log.sql created"
